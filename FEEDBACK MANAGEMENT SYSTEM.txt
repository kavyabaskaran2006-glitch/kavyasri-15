import java.util.ArrayList;
import java.util.Scanner;

// Feedback class
class Feedback {
    private String name;
    private String email;
    private String message;
    private int rating;
    private String comment;

    public Feedback(String name, String email, String message, int rating) {
        this.name = name;
        this.email = email;
        this.message = message;
        this.rating = rating;
        this.comment = generateComment(rating);
    }

    private String generateComment(int rating) {
        switch (rating) {
            case 5: return "Excellent experience!";
            case 4: return "Good service, can be improved.";
            case 3: return "Average experience.";
            case 2: return "Needs improvement.";
            case 1: return "Poor experience.";
            default: return "Invalid rating.";
        }
    }

    public int getRating() {
        return rating;
    }

    public void displayFeedback() {
        System.out.println("--------------------------------------------------");
        System.out.println("Name     : " + name);
        System.out.println("Email    : " + email);
        System.out.println("Rating   : " + rating + " / 5");
        System.out.println("Feedback : " + message);
        System.out.println("Comment  : " + comment);
        System.out.println("--------------------------------------------------");
    }
}

// Main class
public class FeedbackManagementSystem {
    private static ArrayList<Feedback> feedbackList = new ArrayList<>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int roleChoice;

        do {
            System.out.println("\n===== WELCOME TO FEEDBACK MANAGEMENT SYSTEM =====");
            System.out.println("1. Student Login");
            System.out.println("2. Admin Login");
            System.out.println("3. Exit");
            System.out.print("Enter your choice: ");
            roleChoice = sc.nextInt();
            sc.nextLine();

            switch (roleChoice) {
                case 1:
                    studentMenu(sc);
                    break;
                case 2:
                    adminLogin(sc);
                    break;
                case 3:
                    System.out.println(" Exiting system. Goodbye!");
                    break;
                default:
                    System.out.println(" Invalid choice! Please try again.");
            }
        } while (roleChoice != 3);

        sc.close();
    }

    // Student Menu
    private static void studentMenu(Scanner sc) {
        System.out.println("\n Welcome Student!");
        int choice;
        do {
            System.out.println("\n------ Student Menu ------");
            System.out.println("1. Give Feedback");
            System.out.println("2. Back to Main Menu");
            System.out.print("Enter your choice: ");
            choice = sc.nextInt();
            sc.nextLine();

            if (choice == 1) {
                System.out.print("Enter your name: ");
                String name = sc.nextLine();

                System.out.print("Enter your email: ");
                String email = sc.nextLine();

                System.out.print("Enter your feedback message: ");
                String message = sc.nextLine();

                System.out.print("Enter rating (1-5): ");
                int rating = sc.nextInt();
                sc.nextLine();

                feedbackList.add(new Feedback(name, email, message, rating));
                System.out.println(" Thank you for your feedback!");
            }
        } while (choice != 2);
    }

    // Admin Login
    private static void adminLogin(Scanner sc) {
        System.out.print("\n Enter admin password: ");
        String password = sc.nextLine();

        if (password.equals("admin123")) {
            adminMenu(sc);
        } else {
            System.out.println(" Incorrect password! Access denied.");
        }
    }

    // Admin Menu
    private static void adminMenu(Scanner sc) {
        int choice;
        do {
            System.out.println("\n------ Admin Dashboard ------");
            System.out.println("1. View All Feedback");
            System.out.println("2. View Summary Report");
            System.out.println("3. Back to Main Menu");
            System.out.print("Enter your choice: ");
            choice = sc.nextInt();
            sc.nextLine();

            switch (choice) {
                case 1:
                    viewAllFeedback();
                    break;
                case 2:
                    showReport();
                    break;
                case 3:
                    System.out.println("Returning to Main Menu...");
                    break;
                default:
                    System.out.println(" Invalid choice!");
            }
        } while (choice != 3);
    }

    // Display all feedback
    private static void viewAllFeedback() {
        if (feedbackList.isEmpty()) {
            System.out.println(" No feedback available yet.");
        } else {
            System.out.println("\n--- All Feedbacks ---");
            for (Feedback f : feedbackList) {
                f.displayFeedback();
            }
        }
    }

    // Show overall summary report
    private static void showReport() {
        if (feedbackList.isEmpty()) {
            System.out.println(" No feedback to analyze.");
            return;
        }

        int total = 0, max = Integer.MIN_VALUE, min = Integer.MAX_VALUE;
        for (Feedback f : feedbackList) {
            int r = f.getRating();
            total += r;
            if (r > max) max = r;
            if (r < min) min = r;
        }

        double avg = (double) total / feedbackList.size();
        System.out.println(" Average Rating: "+ avg);
        System.out.println(" Highest Rating: " + max);
        System.out.println(" Lowest Rating : " + min);
        System.out.println(" Overall Comment: " + overallComment(avg));
    }

    // Generate overall comment
    private static String overallComment(double avg) {
        if (avg >= 4.5) return "Excellent overall feedback!";
        else if (avg >= 3.5) return "Good overall feedback.";
        else if (avg >= 2.5) return "Average overall feedback.";
        else if (avg >= 1.5) return "Needs improvement overall.";
        else return "Poor overall feedback.";
    }
}
