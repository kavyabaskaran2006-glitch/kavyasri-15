public class Anagram{
    public static void main(String[] args){
        int c=0;
        String str1="listen";
        String str2="silent";
        String a,b;
        a=str1.toLowerCase();
        b=str2.toLowerCase();
        for(int i=0;i<a.length();i++){
            for(int j=0;j<b.length();j++){
                if(a.charAt(i)==b.charAt(j)){
                    c++;
                    break;
                }
            }
        }
        int d=a.length();
        if(c==d){
            System.out.println("Anagram");
        }
        else{
            System.out.println("Not Anagram");
        }
    }
}
